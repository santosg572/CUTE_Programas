# solucion de la ecuacion x'(t) = -x(t-T) para t >= 0
# con x(t)=1 para t en [-T,0]


t2=5

#======================= solucion analitica con retardo

tao = 2
del1 = .1
n = tao/del1+1

t = seq(0,1,length.out = 100)

nrep = 8

t2 = c()
x2 = c()

for (n in 1:nrep){
#n = 2
tt = tao*t+(n-1)*tao
x = 1
for (k in 1:n){
x = x + (-1)^k * (tt-(k-1)*tao)^k / factorial(k)
}
t2 = c(t2, tt)
x2 = c(x2, x)
}

file = 'ecu2_1'
png(paste(file,".png",sep=''))

#print(length(t2))
